# 拖动间隔（秒）：每个点之间的停顿时间，控制拖动速度
#   0.1 = 快速   0.2 = 正常   0.5 = 慢速
拖动间隔 = 0.1

# 点击偏移：匹配到的位置默认是图片中心
#   中心    = 不做偏移（默认）
#   左上角  = 朝左上偏移（往左上角点）
#   右下角  = 朝右下偏移
#   下端    = 朝正下方偏移（适合按钮偏下的情况）
点击偏移 = 中心

# 偏移像素值：配合点击偏移使用的偏移量（像素）
#   数值越大偏得越远
偏移像素 = 15

# 最大处理上限：最多处理多少个目标
#   0 = 不限制
最大上限 = 0
